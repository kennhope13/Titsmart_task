function Ask-Kira {
    param([string]$Text)
    if ([string]::IsNullOrWhiteSpace($Text)) { Write-Host "Vui lòng nhập câu hỏi!" -ForegroundColor Red; return }
    
    $apiKey = "kira_c9b30fdb537ceb426825dd4cef48657e"
    $headers = @{ "Authorization" = "Bearer $apiKey"; "Content-Type" = "application/json; charset=utf-8" }
    $body = @{ 
        model = "kira-mini-1.0"
        messages = @(@{ role = "user"; content = $Text }) 
    } | ConvertTo-Json -Depth 10

    try {
        Write-Host "Đang gọi Kira AI..." -ForegroundColor DarkGray
        $result = Invoke-RestMethod -Uri "https://kiraai.vn/api/v1/chat/completions" -Method Post -Headers $headers -Body $body
        Write-Host "`nKira: $($result.choices[0].message.content)`n" -ForegroundColor Green
    } catch {
        Write-Host "Lỗi: $($_.Exception.Message)" -ForegroundColor Red
    }
}
Set-Alias -Name kira -Value Ask-Kira
